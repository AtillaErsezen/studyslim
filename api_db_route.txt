import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/db';
import { tutors, user, universities, courses, tutorCourses } from '@/db/schema';
import { eq, and, gte, sql, desc } from 'drizzle-orm';

export const dynamic = 'force-dynamic';
export const runtime = 'nodejs';

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    
    if (!body) {
      return NextResponse.json(
        { success: false, error: 'Invalid request body' },
        { status: 400 }
      );
    }

    const { query_type, params } = body;

    if (!query_type) {
      return NextResponse.json(
        { success: false, error: 'Missing query_type' },
        { status: 400 }
      );
    }

    const safeParams = params || {};
    
    let results;
    
    switch (query_type) {
      case 'universal_search':
        const {
          university_short_name,
          course_name,
          department,
          language,
          min_rating,
          max_rate,
        } = safeParams;
        
        const conditions = [
          eq(tutors.verified, true),
          eq(tutors.isAvailable, true),
        ];

        // Fix: use && not ||
        if (university_short_name && university_short_name !== '') {
          conditions.push(eq(universities.shortName, university_short_name));
        }

        if (typeof min_rating === 'number') {
          conditions.push(gte(tutors.ratingAvg, min_rating));
        }

        if (typeof max_rate === 'number') {
          conditions.push(sql`${tutors.hourlyRate} <= ${max_rate}`);
        }

        if (language && language !== '') {
          conditions.push(sql`json_extract(${tutors.languages}, '$') LIKE ${`%${language}%`}`);
        }

        if (department && department !== '') {
          conditions.push(sql`json_extract(${tutors.courseTags}, '$') LIKE ${`%${department}%`}`);
        }

        if (course_name && course_name !== '') {
          conditions.push(sql`LOWER(${courses.name}) LIKE LOWER(${`%${course_name}%`})`);
          
          results = await db
            .select({
              name: user.name,
              id: tutors.id,
              bio: tutors.bio,
              hourlyRate: tutors.hourlyRate,
              ratingAvg: tutors.ratingAvg,
              reviewCount: tutors.reviewCount,
              languages: tutors.languages,
              courseTags: tutors.courseTags,
              verified: tutors.verified,
              isAvailable: tutors.isAvailable,
              universityName: universities.name,
              universityShortName: universities.shortName,
              image: user.image
            })
            .from(tutors)
            .leftJoin(user, eq(tutors.userId, user.id))
            .leftJoin(universities, eq(tutors.universityId, universities.id))
            .leftJoin(tutorCourses, eq(tutors.id, tutorCourses.tutorId))
            .leftJoin(courses, eq(tutorCourses.courseId, courses.id))
            .where(and(...conditions))
            .groupBy(tutors.id)
            .orderBy(desc(tutors.ratingAvg))
            .limit(3);
        } else {
          results = await db
            .select({
              name: user.name,
              id: tutors.id,
              bio: tutors.bio,
              hourlyRate: tutors.hourlyRate,
              ratingAvg: tutors.ratingAvg,
              reviewCount: tutors.reviewCount,
              languages: tutors.languages,
              courseTags: tutors.courseTags,
              verified: tutors.verified,
              isAvailable: tutors.isAvailable,
              universityName: universities.name,
              universityShortName: universities.shortName,
              image: user.image
            })
            .from(tutors)
            .leftJoin(user, eq(tutors.userId, user.id))
            .leftJoin(universities, eq(tutors.universityId, universities.id))
            .where(and(...conditions))
            .groupBy(tutors.id)
            .orderBy(desc(tutors.ratingAvg))
            .limit(3);
        }
        break;

      case 'get_tutor_details':
        if (!safeParams.tutor_id) {
          return NextResponse.json(
            { success: false, error: 'Missing tutor_id' },
            { status: 400 }
          );
        }
        
        results = await db
          .select()
          .from(tutors)
          .where(eq(tutors.id, safeParams.tutor_id))
          .limit(1);
        break;

      default:
        return NextResponse.json(
          { success: false, error: 'Invalid query type' },
          { status: 400 }
        );
    }

    return NextResponse.json({
      success: true,
      data: results,
    });
  } catch (error) {
    console.error('Database query error:', error);
    return NextResponse.json(
      { success: false, error: 'Database query failed'},
      { status: 500 }
    );
  }
}
